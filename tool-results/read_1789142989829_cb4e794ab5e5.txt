    67→    67→export function PlanningView() {
    68→    68→  const [planTab, setPlanTab] = useState<'weekly' | 'monthly'>('weekly')
    69→    69→  // Custom plans (from Plans view wizard) + which plan is being viewed
    70→    70→  const [customPlans, setCustomPlans] = useState<CustomPlan[]>([])
    71→    71→  const [selectedPlanId, setSelectedPlanId] = useState<string | null>(null)
    72→    72→  const storeActivePlanId = useAppStore((s) => s.activePlanId)
    73→    73→  const setActivePlanId = useAppStore((s) => s.setActivePlanId)
    74→    74→
    75→    75→  const basePlan = plans.find((p) => p.type === planTab) || plans[0]
    76→    76→  const activePlan: SimPlan = useMemo(() => {
    77→    77→    if (selectedPlanId) {
    78→    78→      const custom = customPlans.find((p) => p.id === selectedPlanId)
    79→    79→      if (custom) return custom
    80→    80→    }
    81→    81→    return basePlan
    82→    82→  }, [selectedPlanId, customPlans, basePlan])
    83→    83→  const isCustomPlanActive = (activePlan as CustomPlan).isCustom === true
    84→    84→
    85→    85→  const planStart = parseISO(activePlan.startDate)
    86→    86→  const [selectedDate, setSelectedDate] = useState(basePlan.startDate)
    87→    87→  const [sectionFilter, setSectionFilter] = useState<string>('all')
    88→    88→  const [deptFilter, setDeptFilter] = useState<string>('all')
    89→    89→  const [selectedBlockId, setSelectedBlockId] = useState<string | null>(null)
    90→    90→  const [isOptimizing, setIsOptimizing] = useState(false)
    91→    91→  const [optimizationDone, setOptimizationDone] = useState(false)
    92→    92→  const [manualFormOpen, setManualFormOpen] = useState(false)
    93→    93→  const [detailSheetOpen, setDetailSheetOpen] = useState(false)
    94→    94→  const [trackDataVisible, setTrackDataVisible] = useState(false)
    95→    95→  const [timelineMode, setTimelineMode] = useState<'timeline' | 'gantt' | 'crew'>('timeline')
    96→    96→  const [optimizerVisible, setOptimizerVisible] = useState(false)
    97→    97→  const [localBlockDurations, setLocalBlockDurations] = useState<Record<string, number>>({})
    98→    98→  const [localBlockStatuses, setLocalBlockStatuses] = useState<Record<string, SimBlock['status']>>({})
    99→    99→  const [customBlocks, setCustomBlocks] = useState<SimBlock[]>([])
   100→   100→  const isMobile = useIsMobile()
   101→   101→
   102→   102→  // Hydrate from server (source of truth) with one-time localStorage migration
   103→   103→  useEffect(() => {
   104→   104→    let cancelled = false
   105→   105→    const t = setTimeout(async () => {
   106→   106→      try {
   107→   107→        const [mergedBlocks, mergedPlans] = await Promise.all([
   108→   108→          migrateLocalManualBlocks(loadManualBlocks()),
   109→   109→          migrateLocalCustomPlans(loadCustomPlans()),
   110→   110→        ])
   111→   111→        if (cancelled) return
   112→   112→        if (mergedBlocks.length > 0) {
   113→   113→          setCustomBlocks(mergedBlocks)
   114→   114→          persistManualBlocks(mergedBlocks)
   115→   115→        }
   116→   116→        if (mergedPlans.length > 0) setCustomPlans(mergedPlans)
   117→   117→      } catch {
   118→   118→        // server unavailable — localStorage values (loaded above) remain
   119→   119→      }
   120→   120→    }, 0)
   121→   121→    return () => {
   122→   122→      cancelled = true
   123→   123→      clearTimeout(t)
   124→   124→    }
   125→   125→  }, [])
   126→   126→
   127→   127→  // Consume "Open in Planning" requests from the Plans view
   128→   128→  useEffect(() => {
   129→   129→    if (!storeActivePlanId) return
   130→   130→    const t = setTimeout(() => {
   131→   131→      const target = loadCustomPlans().find((p) => p.id === storeActivePlanId)
   132→   132→      if (target) {
   133→   133→        setCustomPlans((prev) => (prev.some((p) => p.id === target.id) ? prev : [...prev, target]))
   134→   134→        setSelectedPlanId(target.id)
   135→   135→        toast.info(`Viewing "${target.name}"`, {
   136→   136→          description: 'Blocks you add are linked to this custom plan',
   137→   137→        })
   138→   138→      }
   139→   139→      setActivePlanId(null)
   140→   140→    }, 0)
   141→   141→    return () => clearTimeout(t)
   142→   142→  }, [storeActivePlanId, setActivePlanId])
   143→   143→
   144→   144→  // Clamp the selected day into the active plan window when switching plans
   145→   145→  useEffect(() => {
   146→   146→    const t = setTimeout(() => {
   147→   147→      setSelectedDate((cur) =>
   148→   148→        cur < activePlan.startDate || cur > activePlan.endDate ? activePlan.startDate : cur
   149→   149→      )
   150→   150→    }, 0)
   151→   151→    return () => clearTimeout(t)
   152→   152→  }, [activePlan.startDate, activePlan.endDate])
   153→   153→
   154→   154→  // Merge simulated blocks with locally created blocks + apply local overrides
   155→   155→  const allBlocks = useMemo(() => (
   156→   156→    [...blocks, ...customBlocks].map((b) => ({
   157→   157→      ...b,
   158→   158→      duration: localBlockDurations[b.id] ?? b.duration,
   159→   159→      status: localBlockStatuses[b.id] ?? b.status,
   160→   160→    }))
   161→   161→  ), [customBlocks, localBlockDurations, localBlockStatuses])
   162→   162→
   163→   163→  const rawSelectedBlock = allBlocks.find((b) => b.id === selectedBlockId) || null
   164→   164→  const selectedBlock = rawSelectedBlock
   165→   165→
   166→   166→  // Deep link from command palette (select block)
   167→   167→  useDeepLink((type, id) => {
   168→   168→    if (type === 'block' && id) {
   169→   169→      const block = allBlocks.find((b) => b.id === id)
   170→   170→      if (block) {
   171→   171→        setSelectedBlockId(block.id)
   172→   172→        if (isMobile) setDetailSheetOpen(true)
   173→   173→      }
   174→   174→    }
   175→   175→  })
   176→   176→  const currentDate = parseISO(selectedDate)
   177→   177→  const goPrevDay = () => {
   178→   178→    const prev = subDays(currentDate, 1)
   179→   179→    if (prev >= planStart) setSelectedDate(format(prev, 'yyyy-MM-dd'))
   180→   180→  }
   181→   181→  const goNextDay = () => {
   182→   182→    const next = addDays(currentDate, 1)
   183→   183→    const planEnd = parseISO(activePlan.endDate)
   184→   184→    if (next <= planEnd) setSelectedDate(format(next, 'yyyy-MM-dd'))
   185→   185→  }
   186→   186→
   187→   187→  const runOptimization = async () => {
   188→   188→    setIsOptimizing(true)
   189→   189→    setOptimizationDone(false)
   190→   190→    try {
   191→   191→      const res = await fetch('/api/optimization', {
   192→   192→        method: 'POST',
   193→   193→        headers: { 'Content-Type': 'application/json' },
   194→   194→        body: JSON.stringify({ planId: activePlan.id, blockIds: activePlan.blockIds }),
   195→   195→      })
   196→   196→      if (!res.ok) throw new Error('Optimization failed')
   197→   197→    } catch {
   198→   198→      // Error handled in optimization progress
   199→   199→    }
   200→   200→  }
   201→   201→
   202→   202→  const onOptimizationComplete = useCallback(() => {
   203→   203→    setIsOptimizing(false)
   204→   204→    setOptimizationDone(true)
   205→   205→  }, [])
   206→   206→
   207→   207→  const handleSelectBlock = (id: string | null) => {
   208→   208→    setSelectedBlockId(id)
   209→   209→    if (id && isMobile) {
   210→   210→      setDetailSheetOpen(true)
   211→   211→    }
   212→   212→    if (!id) {
   213→   213→      setOptimizerVisible(false)
   214→   214→    }
   215→   215→  }
   216→   216→
   217→   217→  const handleApplyDuration = (newDuration: number) => {
   218→   218→    if (selectedBlockId) {
   219→   219→      setLocalBlockDurations(prev => ({ ...prev, [selectedBlockId]: newDuration }))
   220→   220→      setOptimizerVisible(false)
   221→   221→      toast.success('Block duration updated')
   222→   222→    }
   223→   223→  }
   224→   224→
   225→   225→  const handleBlockStatusChange = (blockId: string, nextStatus: SimBlock['status']) => {
   226→   226→    setLocalBlockStatuses(prev => ({ ...prev, [blockId]: nextStatus }))
   227→   227→    const LABELS: Record<string, string> = {
   228→   228→      edited: 'Block edited — submitted for verification',
   229→   229→      verified: 'Block verified',
   230→   230→      finalized: 'Block finalized — ready for approval',
   231→   231→      approved: 'Block approved',
   232→   232→      rejected: 'Block rejected',
   233→   233→    }
   234→   234→    if (nextStatus === 'rejected') {
   235→   235→      toast.error('Block rejected and removed from the recommended workflow')
   236→   236→    } else {
   237→   237→      toast.success(LABELS[nextStatus] || `Block status updated to ${nextStatus}`)
   238→   238→    }
   239→   239→  }
   240→   240→
   241→   241→  const handleCreateManualBlock = (data: {
   242→   242→    name: string
   243→   243→    section: string
   244→   244→    stationFrom: string
   245→   245→    stationTo: string
   246→   246→    startTime: string
   247→   247→    endTime: string
   248→   248→    department: string
   249→   249→    line: string
   250→   250→  }) => {
   251→   251→    // startTime/endTime are "HH:mm" — combine with the selected planning date
   252→   252→    const [sh, sm] = data.startTime.split(':').map(Number)
   253→   253→    const [eh, em] = data.endTime.split(':').map(Number)
   254→   254→    let durationMin = (eh * 60 + em) - (sh * 60 + sm)
   255→   255→    if (durationMin <= 0) durationMin += 24 * 60 // handle overnight blocks
   256→   256→    const startISO = `${selectedDate}T${data.startTime.padStart(5, '0')}:00`
   257→   257→    const endISO = `${selectedDate}T${data.endTime.padStart(5, '0')}:00`
   258→   258→    const newBlock: SimBlock = {
   259→   259→      id: `custom-${Date.now()}`,
   260→   260→      name: data.name,
   261→   261→      section: data.section,
   262→   262→      stationFrom: data.stationFrom,
   263→   263→      stationTo: data.stationTo,
   264→   264→      startTime: startISO,
   265→   265→      endTime: endISO,
   266→   266→      duration: durationMin,
   267→   267→      department: data.department,
   268→   268→      status: 'edited',
   269→   269→      planId: activePlan.id,
   270→   270→      line: (data.line as SimBlock['line']) || 'both',
   271→   271→      isAiRecommended: false,
   272→   272→      aiConfidence: 0,
   273→   273→      aiReasoning: '',
   274→   274→      maintenanceReqIds: [],
   275→   275→    }
   276→   276→    setCustomBlocks(prev => {
   277→   277→      const next = [...prev, newBlock]
   278→   278→      persistManualBlocks(next as (SimBlock & { isManual: true })[])
   279→   279→      return next
   280→   280→    })
   281→   281→    // Link the block to the active custom plan (persisted + server)
   282→   282→    if (isCustomPlanActive) {
   283→   283→      setCustomPlans(addBlockToCustomPlan(activePlan.id, newBlock.id))
   284→   284→      void linkServerCustomPlanBlock(activePlan.id, newBlock.id)
   285→   285→    }
   286→   286→    // Write-through to the server (non-blocking)
   287→   287→    void createServerManualBlock(newBlock as (SimBlock & { isManual: true }))
   288→   288→    toast.success(`Block "${data.name}" added to the plan`, {
   289→   289→      description: `${data.section} · ${durationMin} min · ${format(parseISO(startISO), 'MMM d, HH:mm')}`,
   290→   290→    })
   291→   291→    handleSelectBlock(newBlock.id)
   292→   292→  }
   293→   293→
   294→   294→  // Effective plan block ids — applies the section/department filters + manual blocks
   295→   295→  const effectivePlanBlockIds = useMemo(() => {
   296→   296→    let ids = activePlan.blockIds.filter((id) => {
   297→   297→      const b = allBlocks.find((blk) => blk.id === id)
   298→   298→      if (!b) return false
   299→   299→      if (sectionFilter !== 'all' && b.section !== sectionFilter) return false
   300→   300→      if (deptFilter !== 'all' && b.department !== deptFilter) return false
   301→   301→      return true
   302→   302→    })
   303→   303→    // Manual blocks belong to the plan they were created under
   304→   304→    ids = [...ids, ...customBlocks.filter((b) => b.planId === activePlan.id).map((b) => b.id)]
   305→   305→    return ids
   306→   306→  }, [activePlan, allBlocks, sectionFilter, deptFilter, customBlocks])
   307→   307→
   308→   308→  const planBlocks = allBlocks.filter((b) => effectivePlanBlockIds.includes(b.id))
   309→   309→  const totalBlocks = planBlocks.length
   310→   310→  const aiRecommended = planBlocks.filter((b) => b.isAiRecommended).length
   311→   311→  const planConflicts = conflicts.filter((c) => !c.resolved)
   312→   312→  const criticalConflicts = planConflicts.filter((c) => c.severity === 'critical').length
   313→   313→  const totalHours = planBlocks.reduce((sum, b) => sum + b.duration, 0) / 60
   314→   314→  const departmentsInvolved = new Set(planBlocks.map((b) => b.department)).size
   315→   315→
   316→   316→  const dayLabels: { date: string; label: string; dayNum: number; hasBlocks: boolean; isToday: boolean }[] = []
   317→   317→  const planEnd = parseISO(activePlan.endDate)
   318→   318→  let d = planStart
   319→   319→  while (d <= planEnd) {
   320→   320→    const dateStr = format(d, 'yyyy-MM-dd')
   321→   321→    const hasBlocks = planBlocks.some((b) => {
   322→   322→      const bDate = new Date(b.startTime).toISOString().split('T')[0]
   323→   323→      return bDate === dateStr
   324→   324→    })
   325→   325→    dayLabels.push({ date: dateStr, label: format(d, 'EEE'), dayNum: d.getDate(), hasBlocks, isToday: isToday(d) })
   326→   326→    d = addDays(d, 1)
   327→   327→  }
   328→   328→
   329→   329→  // Detail panel content
   330→   330→  const detailContent = selectedBlock ? (
   331→   331→    <div className="p-4 space-y-3">
   332→   332→      {/* Duration Optimizer toggle */}
   333→   333→      <div className="flex items-center gap-2">
   334→   334→        <Button
   335→   335→          variant={optimizerVisible ? 'default' : 'outline'}
   336→   336→          size="sm"
   337→   337→          className={cn(
   338→   338→            'text-[10px] h-7 px-2.5 gap-1.5',
   339→   339→            optimizerVisible ? 'bg-[#283593] hover:bg-[#0d47a1] text-white' : 'text-[#283593] border-[#9fa8da] hover:bg-[#e8eaf6] dark:text-[#3f51b5] dark:border-[#1a237e] dark:hover:bg-[#0d1442]/30',
   340→   340→          )}
   341→   341→          onClick={() => setOptimizerVisible(!optimizerVisible)}
   342→   342→        >
   343→   343→          <Brain className="w-3.5 h-3.5" />
   344→   344→          Duration Optimizer
   345→   345→        </Button>
   346→   346→      </div>
   347→   347→
   348→   348→      {/* Duration Optimizer panel */}
   349→   349→      {optimizerVisible && (
   350→   350→        <BlockDurationOptimizer
   351→   351→          blockId={selectedBlockId}
   352→   352→          currentDuration={selectedBlock.duration}
   353→   353→          onApply={handleApplyDuration}
   354→   354→        />
   355→   355→      )}
   356→   356→
   357→   357→      {selectedBlock.isAiRecommended ? (
   358→   358→        <>
   359→   359→          <AiRecommendationPanel
   360→   360→            block={selectedBlock}
   361→   361→            onAccept={() => { handleBlockStatusChange(selectedBlock.id, 'edited'); setSelectedBlockId(null); setDetailSheetOpen(false) }}
   362→   362→            onEdit={() => setManualFormOpen(true)}
   363→   363→            onReject={() => { handleBlockStatusChange(selectedBlock.id, 'rejected'); setSelectedBlockId(null); setDetailSheetOpen(false) }}
   364→   364→          />
   365→   365→          <BlockDetailPanel
   366→   366→            block={selectedBlock}
   367→   367→            onEdit={() => setManualFormOpen(true)}
   368→   368→            onClose={() => { setSelectedBlockId(null); setDetailSheetOpen(false) }}
   369→   369→            onReject={() => handleBlockStatusChange(selectedBlock.id, 'rejected')}
   370→   370→            onAdvance={(next) => handleBlockStatusChange(selectedBlock.id, next)}
   371→   371→          />
   372→   372→        </>
   373→   373→      ) : (
   374→   374→        <BlockDetailPanel
   375→   375→          block={selectedBlock}
   376→   376→          onEdit={() => setManualFormOpen(true)}
   377→   377→          onClose={() => { setSelectedBlockId(null); setDetailSheetOpen(false) }}
   378→   378→          onReject={() => handleBlockStatusChange(selectedBlock.id, 'rejected')}
   379→   379→          onAdvance={(next) => handleBlockStatusChange(selectedBlock.id, next)}
   380→   380→        />
   381→   381→      )}
   382→   382→    </div>
   383→   383→  ) : null
   384→   384→
   385→   385→  return (
   386→   386→    <div className="flex flex-col h-full print-area">
   387→   387→      {/* Print Header - hidden on screen, visible when printing */}
   388→   388→      <PrintHeader
   389→   389→        title="RailOpt AI — Block Planning Schedule"
   390→   390→        corridor={corridors[0]?.name ?? 'Railway Corridor'}
   391→   391→        planName={activePlan.name}
   392→   392→      />
   393→   393→
   394→   394→      {/* Header */}
   395→   395→      <div className="flex items-center justify-between gap-4 px-4 sm:px-6 pt-4 pb-2">
   396→   396→        <div className="flex items-center gap-3">
   397→   397→          <CalendarClock className="h-5 w-5 text-[#283593]" />
   398→   398→          <h1 className="text-lg font-semibold text-foreground">Planning</h1>
   399→   399→        </div>
   400→   400→        <div className="flex items-center gap-2">
   401→   401→          <Tabs
   402→   402→            value={isCustomPlanActive ? 'custom-active' : planTab}
   403→   403→            onValueChange={(v) => {
   404→   404→              setPlanTab(v as 'weekly' | 'monthly')
   405→   405→              setSelectedPlanId(null)
   406→   406→            }}
   407→   407→          >
   408→   408→            <TabsList className="h-8">
   409→   409→              <TabsTrigger value="weekly" className="text-xs px-3 h-7">Weekly</TabsTrigger>
   410→   410→              <TabsTrigger value="monthly" className="text-xs px-3 h-7">Monthly</TabsTrigger>
   411→   411→            </TabsList>
   412→   412→          </Tabs>
   413→   413→          {/* Print Schedule button */}
   414→   414→          <Button
   415→   415→            variant="outline"
   416→   416→            size="sm"
   417→   417→            className="text-xs h-8 min-h-[44px] sm:h-8 gap-1.5"
   418→   418→            onClick={() => window.print()}
