    67→export function PlanningView() {
    68→  const [planTab, setPlanTab] = useState<'weekly' | 'monthly'>('weekly')
    69→  // Custom plans (from Plans view wizard) + which plan is being viewed
    70→  const [customPlans, setCustomPlans] = useState<CustomPlan[]>([])
    71→  const [selectedPlanId, setSelectedPlanId] = useState<string | null>(null)
    72→  const storeActivePlanId = useAppStore((s) => s.activePlanId)
    73→  const setActivePlanId = useAppStore((s) => s.setActivePlanId)
    74→
    75→  const basePlan = plans.find((p) => p.type === planTab) || plans[0]
    76→  const activePlan: SimPlan = useMemo(() => {
    77→    if (selectedPlanId) {
    78→      const custom = customPlans.find((p) => p.id === selectedPlanId)
    79→      if (custom) return custom
    80→    }
    81→    return basePlan
    82→  }, [selectedPlanId, customPlans, basePlan])
    83→  const isCustomPlanActive = (activePlan as CustomPlan).isCustom === true
    84→
    85→  const planStart = parseISO(activePlan.startDate)
    86→  const [selectedDate, setSelectedDate] = useState(basePlan.startDate)
    87→  const [sectionFilter, setSectionFilter] = useState<string>('all')
    88→  const [deptFilter, setDeptFilter] = useState<string>('all')
    89→  const [selectedBlockId, setSelectedBlockId] = useState<string | null>(null)
    90→  const [isOptimizing, setIsOptimizing] = useState(false)
    91→  const [optimizationDone, setOptimizationDone] = useState(false)
    92→  const [manualFormOpen, setManualFormOpen] = useState(false)
    93→  const [detailSheetOpen, setDetailSheetOpen] = useState(false)
    94→  const [trackDataVisible, setTrackDataVisible] = useState(false)
    95→  const [timelineMode, setTimelineMode] = useState<'timeline' | 'gantt' | 'crew'>('timeline')
    96→  const [optimizerVisible, setOptimizerVisible] = useState(false)
    97→  const [localBlockDurations, setLocalBlockDurations] = useState<Record<string, number>>({})
    98→  const [localBlockStatuses, setLocalBlockStatuses] = useState<Record<string, SimBlock['status']>>({})
    99→  const [customBlocks, setCustomBlocks] = useState<SimBlock[]>([])
   100→  const isMobile = useIsMobile()
   101→
   102→  // Hydrate from server (source of truth) with one-time localStorage migration
   103→  useEffect(() => {
   104→    let cancelled = false
   105→    const t = setTimeout(async () => {
   106→      try {
   107→        const [mergedBlocks, mergedPlans] = await Promise.all([
   108→          migrateLocalManualBlocks(loadManualBlocks()),
   109→          migrateLocalCustomPlans(loadCustomPlans()),
   110→        ])
   111→        if (cancelled) return
   112→        if (mergedBlocks.length > 0) {
   113→          setCustomBlocks(mergedBlocks)
   114→          persistManualBlocks(mergedBlocks)
   115→        }
   116→        if (mergedPlans.length > 0) setCustomPlans(mergedPlans)
   117→      } catch {
   118→        // server unavailable — localStorage values (loaded above) remain
   119→      }
   120→    }, 0)
   121→    return () => {
   122→      cancelled = true
   123→      clearTimeout(t)
   124→    }
   125→  }, [])
   126→
   127→  // Consume "Open in Planning" requests from the Plans view
   128→  useEffect(() => {
   129→    if (!storeActivePlanId) return
   130→    const t = setTimeout(() => {
   131→      const target = loadCustomPlans().find((p) => p.id === storeActivePlanId)
   132→      if (target) {
   133→        setCustomPlans((prev) => (prev.some((p) => p.id === target.id) ? prev : [...prev, target]))
   134→        setSelectedPlanId(target.id)
   135→        toast.info(`Viewing "${target.name}"`, {
   136→          description: 'Blocks you add are linked to this custom plan',
   137→        })
   138→      }
   139→      setActivePlanId(null)
   140→    }, 0)
   141→    return () => clearTimeout(t)
   142→  }, [storeActivePlanId, setActivePlanId])
   143→
   144→  // Clamp the selected day into the active plan window when switching plans
   145→  useEffect(() => {
   146→    const t = setTimeout(() => {
   147→      setSelectedDate((cur) =>
   148→        cur < activePlan.startDate || cur > activePlan.endDate ? activePlan.startDate : cur
   149→      )
   150→    }, 0)
   151→    return () => clearTimeout(t)
   152→  }, [activePlan.startDate, activePlan.endDate])
   153→
   154→  // Merge simulated blocks with locally created blocks + apply local overrides
   155→  const allBlocks = useMemo(() => (
   156→    [...blocks, ...customBlocks].map((b) => ({
   157→      ...b,
   158→      duration: localBlockDurations[b.id] ?? b.duration,
   159→      status: localBlockStatuses[b.id] ?? b.status,
   160→    }))
   161→  ), [customBlocks, localBlockDurations, localBlockStatuses])
   162→
   163→  const rawSelectedBlock = allBlocks.find((b) => b.id === selectedBlockId) || null
   164→  const selectedBlock = rawSelectedBlock
   165→
   166→  // Deep link from command palette (select block)
   167→  useDeepLink((type, id) => {
   168→    if (type === 'block' && id) {
   169→      const block = allBlocks.find((b) => b.id === id)
   170→      if (block) {
   171→        setSelectedBlockId(block.id)
   172→        if (isMobile) setDetailSheetOpen(true)
   173→      }
   174→    }
   175→  })
   176→  const currentDate = parseISO(selectedDate)
   177→  const goPrevDay = () => {
   178→    const prev = subDays(currentDate, 1)
   179→    if (prev >= planStart) setSelectedDate(format(prev, 'yyyy-MM-dd'))
   180→  }
   181→  const goNextDay = () => {
   182→    const next = addDays(currentDate, 1)
   183→    const planEnd = parseISO(activePlan.endDate)
   184→    if (next <= planEnd) setSelectedDate(format(next, 'yyyy-MM-dd'))
   185→  }
   186→
   187→  const runOptimization = async () => {
   188→    setIsOptimizing(true)
   189→    setOptimizationDone(false)
   190→    try {
   191→      const res = await fetch('/api/optimization', {
   192→        method: 'POST',
   193→        headers: { 'Content-Type': 'application/json' },
   194→        body: JSON.stringify({ planId: activePlan.id, blockIds: activePlan.blockIds }),
   195→      })
   196→      if (!res.ok) throw new Error('Optimization failed')
   197→    } catch {
   198→      // Error handled in optimization progress
   199→    }
   200→  }
   201→
   202→  const onOptimizationComplete = useCallback(() => {
   203→    setIsOptimizing(false)
   204→    setOptimizationDone(true)
   205→  }, [])
   206→
   207→  const handleSelectBlock = (id: string | null) => {
   208→    setSelectedBlockId(id)
   209→    if (id && isMobile) {
   210→      setDetailSheetOpen(true)
   211→    }
   212→    if (!id) {
   213→      setOptimizerVisible(false)
   214→    }
   215→  }
   216→
   217→  const handleApplyDuration = (newDuration: number) => {
   218→    if (selectedBlockId) {
   219→      setLocalBlockDurations(prev => ({ ...prev, [selectedBlockId]: newDuration }))
   220→      setOptimizerVisible(false)
   221→      toast.success('Block duration updated')
   222→    }
   223→  }
   224→
   225→  const handleBlockStatusChange = (blockId: string, nextStatus: SimBlock['status']) => {
   226→    setLocalBlockStatuses(prev => ({ ...prev, [blockId]: nextStatus }))
   227→    const LABELS: Record<string, string> = {
   228→      edited: 'Block edited — submitted for verification',
   229→      verified: 'Block verified',
   230→      finalized: 'Block finalized — ready for approval',
   231→      approved: 'Block approved',
   232→      rejected: 'Block rejected',
   233→    }
   234→    if (nextStatus === 'rejected') {
   235→      toast.error('Block rejected and removed from the recommended workflow')
   236→    } else {
   237→      toast.success(LABELS[nextStatus] || `Block status updated to ${nextStatus}`)
   238→    }
   239→  }
   240→
   241→  const handleCreateManualBlock = (data: {
   242→    name: string
   243→    section: string
   244→    stationFrom: string
   245→    stationTo: string
   246→    startTime: string
   247→    endTime: string
   248→    department: string
   249→    line: string
   250→  }) => {
   251→    // startTime/endTime are "HH:mm" — combine with the selected planning date
   252→    const [sh, sm] = data.startTime.split(':').map(Number)
   253→    const [eh, em] = data.endTime.split(':').map(Number)
   254→    let durationMin = (eh * 60 + em) - (sh * 60 + sm)
   255→    if (durationMin <= 0) durationMin += 24 * 60 // handle overnight blocks
   256→    const startISO = `${selectedDate}T${data.startTime.padStart(5, '0')}:00`
   257→    const endISO = `${selectedDate}T${data.endTime.padStart(5, '0')}:00`
   258→    const newBlock: SimBlock = {
   259→      id: `custom-${Date.now()}`,
   260→      name: data.name,
   261→      section: data.section,
   262→      stationFrom: data.stationFrom,
   263→      stationTo: data.stationTo,
   264→      startTime: startISO,
   265→      endTime: endISO,
   266→      duration: durationMin,
   267→      department: data.department,
   268→      status: 'edited',
   269→      planId: activePlan.id,
   270→      line: (data.line as SimBlock['line']) || 'both',
   271→      isAiRecommended: false,
   272→      aiConfidence: 0,
   273→      aiReasoning: '',
   274→      maintenanceReqIds: [],
   275→    }
   276→    setCustomBlocks(prev => {
   277→      const next = [...prev, newBlock]
   278→      persistManualBlocks(next as (SimBlock & { isManual: true })[])
   279→      return next
   280→    })
   281→    // Link the block to the active custom plan (persisted + server)
   282→    if (isCustomPlanActive) {
   283→      setCustomPlans(addBlockToCustomPlan(activePlan.id, newBlock.id))
   284→      void linkServerCustomPlanBlock(activePlan.id, newBlock.id)
   285→    }
   286→    // Write-through to the server (non-blocking)
   287→    void createServerManualBlock(newBlock as (SimBlock & { isManual: true }))
   288→    toast.success(`Block "${data.name}" added to the plan`, {
   289→      description: `${data.section} · ${durationMin} min · ${format(parseISO(startISO), 'MMM d, HH:mm')}`,
   290→    })
   291→    handleSelectBlock(newBlock.id)
   292→  }
   293→
   294→  // Drag-and-drop / keyboard reschedule of a manual block on the timeline.
   295→  // newStartH is snapped to 15 min and clamped to [0, 24h − duration] by the timeline;
   296→  // conflictBlockNames lists same-day blocks the proposed slot overlaps (live warning).
   297→  const handleMoveBlock = useCallback((blockId: string, newStartH: number, conflictBlockNames: string[] = [], source: 'drag' | 'keyboard' = 'drag') => {
   298→    const block = customBlocks.find((b) => b.id === blockId)
   299→    if (!block) return
   300→    const durationMin = Math.max(block.duration, 15)
   301→    const startTotal = Math.round(newStartH * 60)
   302→    const endTotal = Math.min(24 * 60, startTotal + durationMin)
   303→    const pad = (n: number) => n.toString().padStart(2, '0')
   304→    const sh = Math.floor(startTotal / 60)
   305→    const sm = startTotal % 60
   306→    const eh = Math.floor(endTotal / 60)
   307→    const em = endTotal % 60
   308→    const startTime = `${selectedDate}T${pad(sh)}:${pad(sm)}:00`
   309→    const endTime = `${selectedDate}T${pad(eh)}:${pad(em)}:00`
   310→    if (startTime === block.startTime && endTime === block.endTime) return
   311→
   312→    const previous = { startTime: block.startTime, endTime: block.endTime, duration: block.duration }
   313→    const nextTimes = { startTime, endTime, duration: endTotal - startTotal }
   314→
   315→    // localStorage is written on every mutation, so it is always the freshest full list
   316→    const applyTimes = (times: { startTime: string; endTime: string; duration: number }) => {
   317→      setCustomBlocks((prev) => prev.map((b) => (b.id === blockId ? { ...b, ...times } : b)))
   318→      persistManualBlocks(loadManualBlocks().map((b) => (b.id === blockId ? { ...b, ...times } : b)) as (SimBlock & { isManual: true })[])
   319→      void updateServerManualBlock(blockId, times)
   320→    }
   321→    applyTimes(nextTimes)
   322→
   323→    const oldLabel = previous.startTime.split('T')[1]?.slice(0, 5) ?? ''
   324→    const newLabel = `${pad(sh)}:${pad(sm)}`
   325→    const undoAction = {
   326→      label: 'Undo',
   327→      onClick: () => {
   328→        applyTimes(previous)
   329→        toast.success('Reschedule reverted', {
   330→          description: `${block.name} restored to ${oldLabel}`,
   331→        })
   332→      },
   333→    }
   334→    if (conflictBlockNames.length > 0) {
   335→      toast.warning('Block moved — overlaps detected', {
   336→        description: `${block.name}: ${oldLabel} → ${newLabel} · overlaps ${conflictBlockNames.slice(0, 2).join(', ')}${conflictBlockNames.length > 2 ? ` +${conflictBlockNames.length - 2} more` : ''}`,
   337→        action: undoAction,
   338→        duration: 9000,
   339→      })
   340→    } else if (source !== 'keyboard') {
   341→      // Keyboard nudges stay silent on success — the block visibly moves; no toast spam
   342→      toast.success('Block rescheduled', {
   343→        description: `${block.name}: ${oldLabel} → ${newLabel} (saved to server)`,
   344→        action: undoAction,
   345→        duration: 8000,
   346→      })
   347→    }
   348→  }, [customBlocks, selectedDate])
   349→
   350→  // Effective plan block ids — applies the section/department filters + manual blocks
   351→  const effectivePlanBlockIds = useMemo(() => {
   352→    let ids = activePlan.blockIds.filter((id) => {
   353→      const b = allBlocks.find((blk) => blk.id === id)
   354→      if (!b) return false
   355→      if (sectionFilter !== 'all' && b.section !== sectionFilter) return false
   356→      if (deptFilter !== 'all' && b.department !== deptFilter) return false
   357→      return true
   358→    })
   359→    // Manual blocks belong to the plan they were created under
   360→    ids = [...ids, ...customBlocks.filter((b) => b.planId === activePlan.id).map((b) => b.id)]
   361→    return ids
   362→  }, [activePlan, allBlocks, sectionFilter, deptFilter, customBlocks])
   363→
   364→  const planBlocks = allBlocks.filter((b) => effectivePlanBlockIds.includes(b.id))
   365→  const totalBlocks = planBlocks.length
   366→  const aiRecommended = planBlocks.filter((b) => b.isAiRecommended).length
   367→  const planConflicts = conflicts.filter((c) => !c.resolved)
   368→  const criticalConflicts = planConflicts.filter((c) => c.severity === 'critical').length
   369→  const totalHours = planBlocks.reduce((sum, b) => sum + b.duration, 0) / 60
   370→  const departmentsInvolved = new Set(planBlocks.map((b) => b.department)).size
   371→
   372→  const dayLabels: { date: string; label: string; dayNum: number; hasBlocks: boolean; isToday: boolean }[] = []
   373→  const planEnd = parseISO(activePlan.endDate)
   374→  let d = planStart
   375→  while (d <= planEnd) {
   376→    const dateStr = format(d, 'yyyy-MM-dd')
   377→    const hasBlocks = planBlocks.some((b) => {
   378→      const bDate = new Date(b.startTime).toISOString().split('T')[0]
   379→      return bDate === dateStr
   380→    })
   381→    dayLabels.push({ date: dateStr, label: format(d, 'EEE'), dayNum: d.getDate(), hasBlocks, isToday: isToday(d) })
   382→    d = addDays(d, 1)
   383→  }
   384→
   385→  // Detail panel content
   386→  const detailContent = selectedBlock ? (
   387→    <div className="p-4 space-y-3">
   388→      {/* Duration Optimizer toggle */}
   389→      <div className="flex items-center gap-2">
   390→        <Button
   391→          variant={optimizerVisible ? 'default' : 'outline'}
   392→          size="sm"
   393→          className={cn(
   394→            'text-[10px] h-7 px-2.5 gap-1.5',
   395→            optimizerVisible ? 'bg-[#283593] hover:bg-[#0d47a1] text-white' : 'text-[#283593] border-[#9fa8da] hover:bg-[#e8eaf6] dark:text-[#3f51b5] dark:border-[#1a237e] dark:hover:bg-[#0d1442]/30',
   396→          )}
   397→          onClick={() => setOptimizerVisible(!optimizerVisible)}
   398→        >
   399→          <Brain className="w-3.5 h-3.5" />
   400→          Duration Optimizer
   401→        </Button>
   402→      </div>
   403→
   404→      {/* Duration Optimizer panel */}
   405→      {optimizerVisible && (
   406→        <BlockDurationOptimizer
   407→          blockId={selectedBlockId}
   408→          currentDuration={selectedBlock.duration}
   409→          onApply={handleApplyDuration}
   410→        />
   411→      )}
   412→
   413→      {selectedBlock.isAiRecommended ? (
   414→        <>
   415→          <AiRecommendationPanel
   416→            block={selectedBlock}
   417→            onAccept={() => { handleBlockStatusChange(selectedBlock.id, 'edited'); setSelectedBlockId(null); setDetailSheetOpen(false) }}
   418→            onEdit={() => setManualFormOpen(true)}
   419→            onReject={() => { handleBlockStatusChange(selectedBlock.id, 'rejected'); setSelectedBlockId(null); setDetailSheetOpen(false) }}
   420→          />
   421→          <BlockDetailPanel
   422→            block={selectedBlock}
   423→            onEdit={() => setManualFormOpen(true)}
   424→            onClose={() => { setSelectedBlockId(null); setDetailSheetOpen(false) }}
   425→            onReject={() => handleBlockStatusChange(selectedBlock.id, 'rejected')}
   426→            onAdvance={(next) => handleBlockStatusChange(selectedBlock.id, next)}
   427→          />
   428→        </>
   429→      ) : (
   430→        <BlockDetailPanel
   431→          block={selectedBlock}
   432→          onEdit={() => setManualFormOpen(true)}
   433→          onClose={() => { setSelectedBlockId(null); setDetailSheetOpen(false) }}
   434→          onReject={() => handleBlockStatusChange(selectedBlock.id, 'rejected')}
   435→          onAdvance={(next) => handleBlockStatusChange(selectedBlock.id, next)}
   436→        />
   437→      )}
   438→    </div>
   439→  ) : null
   440→
   441→  return (
   442→    <div className="flex flex-col h-full print-area">
   443→      {/* Print Header - hidden on screen, visible when printing */}
   444→      <PrintHeader
   445→        title="RailOpt AI — Block Planning Schedule"
   446→        corridor={corridors[0]?.name ?? 'Railway Corridor'}
   447→        planName={activePlan.name}
   448→      />
   449→
   450→      {/* Header */}
   451→      <div className="flex items-center justify-between gap-4 px-4 sm:px-6 pt-4 pb-2">
   452→        <div className="flex items-center gap-3">
   453→          <CalendarClock className="h-5 w-5 text-[#283593]" />
   454→          <h1 className="text-lg font-semibold text-foreground">Planning</h1>
   455→        </div>
   456→        <div className="flex items-center gap-2">
   457→          <Tabs
   458→            value={isCustomPlanActive ? 'custom-active' : planTab}
   459→            onValueChange={(v) => {
   460→              setPlanTab(v as 'weekly' | 'monthly')
   461→              setSelectedPlanId(null)
   462→            }}
   463→          >
   464→            <TabsList className="h-8">
   465→              <TabsTrigger value="weekly" className="text-xs px-3 h-7">Weekly</TabsTrigger>
   466→              <TabsTrigger value="monthly" className="text-xs px-3 h-7">Monthly</TabsTrigger>
   467→            </TabsList>
   468→          </Tabs>
   469→          {/* Print Schedule button */}
   470→          <Button
   471→            variant="outline"
   472→            size="sm"
   473→            className="text-xs h-8 min-h-[44px] sm:h-8 gap-1.5"
   474→            onClick={() => window.print()}
   475→          >
   476→            <Printer className="h-3.5 w-3.5" />
   477→            <span className="hidden sm:inline">Print Schedule</span>
   478→            <span className="sm:hidden">Print</span>
   479→          </Button>
   480→        </div>
   481→      </div>
   482→
   483→      {/* Plan info strip — plan selector + status */}
   484→      <div className="flex items-center gap-3 px-4 sm:px-6 pb-2 flex-wrap">
   485→        <Select
   486→          value={activePlan.id}
   487→          onValueChange={(id) => {
