    49→/* ─── Animated Counter Component ─── */
    50→function AnimatedCounter({
    51→  end,
    52→  duration = 2000,
    53→  className = '',
    54→  prefix = '',
    55→  suffix = '',
    56→  displayOverride,
    57→}: {
    58→  end: number
    59→  duration?: number
    60→  className?: string
    61→  prefix?: string
    62→  suffix?: string
    63→  displayOverride?: string
    64→}) {
    65→  const [count, setCount] = useState(0)
    66→  const ref = useRef<HTMLDivElement>(null)
    67→  const isInView = useInView(ref, { once: true, margin: '-100px' })
    68→  const shouldReduce = useReducedMotion()
    69→  const hasStarted = useRef(false)
    70→
    71→  useEffect(() => {
    72→    if (!isInView || hasStarted.current) return
    73→    hasStarted.current = true
    74→
    75→    let rafId: number
    76→    if (shouldReduce) {
    77→      rafId = requestAnimationFrame(() => setCount(end))
    78→    } else {
    79→      let startTime: number | null = null
    80→      const step = (timestamp: number) => {
    81→        if (!startTime) startTime = timestamp
    82→        const progress = Math.min((timestamp - startTime) / duration, 1)
    83→        const eased = 1 - Math.pow(1 - progress, 3)
    84→        setCount(Math.floor(eased * end))
    85→        if (progress < 1) {
    86→          rafId = requestAnimationFrame(step)
    87→        }
    88→      }
    89→      rafId = requestAnimationFrame(step)
    90→    }
    91→    return () => cancelAnimationFrame(rafId)
    92→  }, [isInView, end, duration, shouldReduce])
    93→
    94→  return (
    95→    <div ref={ref} className={className}>
    96→      {prefix}
    97→      {displayOverride ? displayOverride : count.toLocaleString('en-IN')}
    98→      {suffix}
    99→    </div>
   100→  )
   101→}

   114→/* ─── Ashoka Chakra Inspired Pattern ─── */
   115→function ChakraPattern() {
   116→  const spokes = 24
   117→  return (
   118→    <div className="absolute right-8 top-1/2 -translate-y-1/2 w-64 h-64 opacity-[0.06] pointer-events-none hidden lg:block" aria-hidden="true">
   119→      <svg viewBox="0 0 200 200" className="w-full h-full">
   120→        <circle cx="100" cy="100" r="95" fill="none" stroke="white" strokeWidth="1.5" />
   121→        <circle cx="100" cy="100" r="75" fill="none" stroke="white" strokeWidth="0.5" />
   122→        <circle cx="100" cy="100" r="20" fill="none" stroke="white" strokeWidth="1" />
   123→        {Array.from({ length: spokes }).map((_, i) => {
   124→          const angle = (i * 360) / spokes
   125→          const rad = (angle * Math.PI) / 180
   126→          const x1 = 100 + 20 * Math.cos(rad)
   127→          const y1 = 100 + 20 * Math.sin(rad)
   128→          const x2 = 100 + 95 * Math.cos(rad)
   129→          const y2 = 100 + 95 * Math.sin(rad)
   130→          return (
   131→            <line key={i} x1={x1} y1={y1} x2={x2} y2={y2} stroke="white" strokeWidth="0.8" />
   132→          )
   133→        })}
   134→      </svg>
   135→    </div>
   136→  )
   137→}

   315→/* ─── Animated Text Reveal ─── */
   316→function AnimatedTitle({ text, className = '' }: { text: string; className?: string }) {
   317→  const shouldReduce = useReducedMotion()
   318→  const ref = useRef<HTMLDivElement>(null)
   319→  const isInView = useInView(ref, { once: true })
   320→
   321→  if (shouldReduce) {
   322→    return <div ref={ref} className={className}>{text}</div>
   323→  }
   324→
   325→  return (
   326→    <div ref={ref} className={className} aria-label={text}>
   327→      {text.split('').map((char, i) => (
   328→        <motion.span
   329→          key={i}
   330→          initial={{ opacity: 0, y: 20 }}
   331→          animate={isInView ? { opacity: 1, y: 0 } : {}}
   332→          transition={{ duration: 0.4, delay: i * 0.05, ease: 'easeOut' }}
   333→          className="inline-block"
   334→          aria-hidden="true"
   335→        >
   336→          {char === ' ' ? '\u00A0' : char}
   337→        </motion.span>
   338→      ))}
   339→    </div>
   340→  )
   341→}
