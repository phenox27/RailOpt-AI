   100→// Circular progress indicator component
   101→function CircularProgress({ value, size = 40, strokeWidth = 3 }: { value: number; size?: number; strokeWidth?: number }) {
   102→  const radius = (size - strokeWidth) / 2
   103→  const circumference = 2 * Math.PI * radius
   104→  const progress = ((100 - value) / 100) * circumference
   105→
   106→  const color = value >= 80 ? 'text-emerald-500' : value >= 60 ? 'text-amber-500' : 'text-red-500'
   107→  const strokeColor = value >= 80 ? '#10b981' : value >= 60 ? '#f59e0b' : '#ef4444'
   108→  const bgColor = value >= 80 ? '#10b98120' : value >= 60 ? '#f59e0b20' : '#ef444420'
   109→
   110→  return (
   111→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   112→      <svg width={size} height={size} className="-rotate-90">
   113→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={bgColor} strokeWidth={strokeWidth} />
   114→        <circle
   115→          cx={size / 2}
   116→          cy={size / 2}
   117→          r={radius}
   118→          fill="none"
   119→          stroke={strokeColor}
   120→          strokeWidth={strokeWidth}
   121→          strokeDasharray={circumference}
   122→          strokeDashoffset={progress}
   123→          strokeLinecap="round"
   124→          className="transition-all duration-700 ease-out"
   125→        />
   126→      </svg>
   127→      <span className={`absolute text-[10px] font-bold ${color}`}>{value}</span>
   128→    </div>
   129→  )
   130→}
