   298→   298→  // Drag-and-drop / keyboard reschedule of a manual block on the timeline.

   300→   300→  // conflictBlockNames lists same-day blocks the proposed slot overlaps (live warning),
   301→   301→  // trainNumbers lists corridor trains whose path crosses the proposed slot.
   302→   302→  const handleMoveBlock = useCallback((blockId: string, newStartH: number, conflictBlockNames: string[] = [], source: 'drag' | 'keyboard' = 'drag', trainNumbers: string[] = []) => {
   303→   303→    const block = customBlocks.find((b) => b.id === blockId)
   304→   304→    if (!block) return
   305→   305→    const durationMin = Math.max(block.duration, 15)
   306→   306→    const startTotal = Math.round(newStartH * 60)
   307→   307→    const endTotal = Math.min(24 * 60, startTotal + durationMin)
   308→   308→    const pad = (n: number) => n.toString().padStart(2, '0')
   309→   309→    const sh = Math.floor(startTotal / 60)

   327→   327→
   328→   328→    const oldLabel = previous.startTime.split('T')[1]?.slice(0, 5) ?? ''
   329→   329→    const newLabel = `${pad(sh)}:${pad(sm)}`
   330→   330→    const undoAction = {
   331→   331→      label: 'Undo',
   332→   332→      onClick: () => {
   333→   333→        applyTimes(previous)
   334→   334→        toast.success('Reschedule reverted', {
   335→   335→          description: `${block.name} restored to ${oldLabel}`,
   336→   336→        })
   337→   337→      },
   338→   338→    }
   339→   339→    const trainSuffix = trainNumbers.length > 0 ? ` · train path: ${trainNumbers.slice(0, 3).join(', ')}${trainNumbers.length > 3 ? ` +${trainNumbers.length - 3}` : ''}` : ''
   340→   340→    if (conflictBlockNames.length > 0) {
   341→   341→      toast.warning('Block moved — overlaps detected', {
   342→   342→        description: `${block.name}: ${oldLabel} → ${newLabel} · overlaps ${conflictBlockNames.slice(0, 2).join(', ')}${conflictBlockNames.length > 2 ? ` +${conflictBlockNames.length - 2} more` : ''}${trainSuffix}`,
   343→   343→        action: undoAction,
   344→   344→        duration: 9000,
   345→   345→      })

   349→   349→        description: `${block.name}: ${oldLabel} → ${newLabel} · crosses train ${trainNumbers.slice(0, 2).join(', ')}${trainNumbers.length > 2 ? ` +${trainNumbers.length - 2}` : ''}`,
   350→   350→        action: undoAction,
   351→   351→        duration: 9000,
   352→   352→      })

   384→   384→  // Click an hour bucket in the Corridor Utilization panel → select that hour's first block
   385→   385→  const handleHourJump = useCallback((hour: number) => {
   386→   386→    const startOf = (b: SimBlock) => {
   387→   387→      const d = new Date(b.startTime)
   388→   388→      return d.getHours() + d.getMinutes() / 60
   389→   389→    }
   390→   390→    const starting = utilDayBlocks
   391→   391→      .filter((b) => startOf(b) >= hour && startOf(b) < hour + 1)
   392→   392→      .sort((a, b) => startOf(a) - startOf(b))
   393→   393→    const covering = utilDayBlocks
   394→   394→      .filter((b) => startOf(b) <= hour && startOf(b) + b.duration / 60 > hour)
   395→   395→      .sort((a, b) => startOf(a) - startOf(b))
   396→   396→    const target = starting[0] ?? covering[0]
   397→   397→    if (target) {
   398→   398→      handleSelectBlock(target.id)
   399→   399→      const dep = new Date(target.startTime).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
