    59→/** A manual block is draggable: DB-loaded blocks carry isManual, localStorage ones use the custom- id prefix. */
    60→export function isMovable(block: SimBlock, canMove: boolean): boolean {
    61→  if (!canMove) return false
    62→  return (block as { isManual?: boolean }).isManual === true || block.id.startsWith('custom-')
    63→}

    65→// Parse ISO time to hours (e.g. "2025-01-27T01:00:00" -> 1.0)
    66→function timeToHours(isoString: string): number {
    67→  const d = new Date(isoString)
    68→  return d.getHours() + d.getMinutes() / 60
    69→}
    70→
    71→// Format hours to HH:MM
    72→function formatHour(h: number): string {
    73→  const hr = Math.floor(h)
    74→  const min = Math.round((h - hr) * 60)
    75→  return `${hr.toString().padStart(2, '0')}:${min.toString().padStart(2, '0')}`
    76→}

    78→// Check if a block belongs to the selected date
    79→function isBlockOnDate(block: SimBlock, dateStr: string): boolean {
    80→  const blockDate = new Date(block.startTime).toISOString().split('T')[0]
    81→  return blockDate === dateStr
    82→}

    84→interface TimelineLane {
    85→  laneIndex: number
    86→  blocks: SimBlock[]
    87→}
    88→
    89→// Simple lane assignment: place blocks into lanes to avoid overlap
    90→function assignLanes(dayBlocks: SimBlock[]): TimelineLane[] {
    91→  const lanes: SimBlock[][] = []
    92→
    93→  for (const block of dayBlocks) {
    94→    const startH = timeToHours(block.startTime)
    95→    const endH = timeToHours(block.endTime)
    96→
    97→    let placed = false
    98→    for (let i = 0; i < lanes.length; i++) {
    99→      const hasOverlap = lanes[i].some((b) => {
   100→        const bs = timeToHours(b.startTime)
   101→        const be = timeToHours(b.endTime)
   102→        return startH < be && endH > bs
   103→      })
   104→      if (!hasOverlap) {
   105→        lanes[i].push(block)
   106→        placed = true
   107→        break
   108→      }
   109→    }
   110→    if (!placed) {
   111→      lanes.push([block])
   112→    }
   113→  }
   114→
   115→  return lanes.map((blocks, idx) => ({ laneIndex: idx, blocks }))
   116→}

   118→/** Names of same-day blocks whose time window overlaps [startH, startH+durationH]. */
   119→export function computeOverlaps(dayBlocks: SimBlock[], draggedId: string, startH: number, durationH: number): string[] {
   120→  const endH = startH + durationH
   121→  return dayBlocks
   122→    .filter((b) => {
   123→      if (b.id === draggedId) return false
   124→      const bs = timeToHours(b.startTime)
   125→      const be = timeToHours(b.endTime)
   126→      return startH < be && endH > bs
   127→    })
   128→    .map((b) => b.name)
   129→}

   131→interface DragState {
   132→  blockId: string
   133→  origStartH: number
   134→  durationH: number
   135→  deltaH: number // snapped shift, may be 0
   136→  active: boolean // passed the movement threshold
   137→}

   139→export function BlockTimeline({ selectedDate, selectedBlockId, onSelectBlock, planBlockIds, extraBlocks, onMoveBlock }: BlockTimelineProps) {
   140→  const [hoveredBlockId, setHoveredBlockId] = useState<string | null>(null)
   141→  const [drag, setDrag] = useState<DragState | null>(null)
   142→  const timelineRef = useRef<HTMLDivElement>(null)
   143→  const dragRef = useRef<{
   144→    blockId: string
   145→    origStartH: number
   146→    durationH: number
   147→    startX: number
   148→    widthPx: number
   149→    moved: boolean
   150→  } | null>(null)
   151→  // Mirrors the latest committed drag state so commit-time side effects
   152→  // (calling onMoveBlock) never run inside a setState updater.
   153→  const dragValueRef = useRef<DragState | null>(null)
   154→  // Suppresses the synthetic click that follows a completed drag pointerup.
   155→  const suppressClickRef = useRef(false)
   156→
   157→  // Filter blocks for the selected day and plan (including locally created blocks)
   158→  const dayBlocks = useMemo(() => {
   159→    const source = extraBlocks && extraBlocks.length > 0 ? [...blocks, ...extraBlocks] : blocks
   160→    const filtered = source.filter((b) => {
   161→      if (planBlockIds && !planBlockIds.includes(b.id)) return false
   162→      return isBlockOnDate(b, selectedDate)
   163→    })
   164→    return filtered.sort((a, b) => timeToHours(a.startTime) - timeToHours(b.startTime))
   165→  }, [selectedDate, planBlockIds, extraBlocks])
   166→
   167→  const lanes = useMemo(() => assignLanes(dayBlocks), [dayBlocks])
   168→
   169→  // Live conflict detection while dragging: same-day blocks the proposed slot overlaps
   170→  const dragConflicts = useMemo(() => {
   171→    if (!drag?.active) return [] as string[]
   172→    return computeOverlaps(dayBlocks, drag.blockId, drag.origStartH + drag.deltaH, drag.durationH)
   173→  }, [drag, dayBlocks])
   174→
   175→  const clampStart = useCallback((startH: number, durationH: number) => {
   176→    return Math.min(Math.max(0, startH), TOTAL_HOURS - durationH)
   177→  }, [])
   178→
   179→  // ---- Drag to reschedule (manual blocks only) ----
   180→
   181→  const handleDragPointerDown = useCallback((e: React.PointerEvent<HTMLButtonElement>, block: SimBlock) => {
   182→    if (!onMoveBlock || e.button !== 0) return
   183→    const laneEl = e.currentTarget.closest('[data-lane]') as HTMLElement | null
   184→    const widthPx = (laneEl ?? e.currentTarget).getBoundingClientRect().width
   185→    if (widthPx <= 0) return
   186→    dragRef.current = {
   187→      blockId: block.id,
   188→      origStartH: timeToHours(block.startTime),
   189→      durationH: Math.max(block.duration, 15) / 60,
   190→      startX: e.clientX,
   191→      widthPx,
   192→      moved: false,
   193→    }
   194→  }, [onMoveBlock])
   195→
   196→  const handleDragPointerMove = useCallback((e: React.PointerEvent<HTMLButtonElement>) => {
   197→    const st = dragRef.current
   198→    if (!st) return
   199→    const dx = e.clientX - st.startX
   200→    if (!st.moved && Math.abs(dx) < DRAG_THRESHOLD_PX) return
   201→    st.moved = true
   202→    const rawDelta = (dx / st.widthPx) * TOTAL_HOURS
   203→    const deltaH = Math.round(rawDelta / SNAP_H) * SNAP_H
   204→    const nextDelta = clampStart(st.origStartH + deltaH, st.durationH) - st.origStartH
   205→    const prev = dragValueRef.current
   206→    if (prev && prev.blockId === st.blockId && prev.deltaH === nextDelta && prev.active) return
   207→    const next: DragState = { blockId: st.blockId, origStartH: st.origStartH, durationH: st.durationH, deltaH: nextDelta, active: true }
   208→    dragValueRef.current = next
   209→    setDrag(next)
   210→  }, [clampStart])
   211→
   212→  const endDrag = useCallback((commit: boolean) => {
   213→    const st = dragRef.current
   214→    const current = dragValueRef.current
   215→    suppressClickRef.current = st?.moved === true
   216→    if (commit && st && current && current.blockId === st.blockId && current.active && current.deltaH !== 0) {
   217→      const newStart = current.origStartH + current.deltaH
   218→      const conflicts = computeOverlaps(dayBlocks, st.blockId, newStart, current.durationH)
   219→      onMoveBlock?.(st.blockId, newStart, conflicts)
   220→    }
   221→    dragValueRef.current = null
   222→    dragRef.current = null
   223→    setDrag(null)
   224→  }, [onMoveBlock, dayBlocks])
   225→
   226→  const handleDragPointerUp = useCallback(() => endDrag(true), [endDrag])
   227→  const handleDragPointerCancel = useCallback(() => endDrag(false), [endDrag])
   228→
   229→  // Keyboard nudge (Shift+←/→ moves a focused manual block by 15 min)
   230→  const handleBlockKeyDown = useCallback((e: React.KeyboardEvent<HTMLButtonElement>, block: SimBlock) => {
   231→    if (!onMoveBlock || !(e.shiftKey && (e.key === 'ArrowLeft' || e.key === 'ArrowRight'))) return
   232→    e.preventDefault()
   233→    e.stopPropagation()
   234→    const durationH = Math.max(block.duration, 15) / 60
   235→    const orig = timeToHours(block.startTime)
   236→    const next = clampStart(orig + (e.key === 'ArrowRight' ? SNAP_H : -SNAP_H), durationH)
   237→    if (next !== orig) {
   238→      const conflicts = computeOverlaps(dayBlocks, block.id, next, durationH)
   239→      onMoveBlock(block.id, next, conflicts, 'keyboard')
   240→    }
   241→  }, [onMoveBlock, clampStart, dayBlocks])
   242→
   243→  // Arrow key navigation for block items
   244→  const handleTimelineKeyDown = useCallback((e: React.KeyboardEvent) => {
   245→    const blockButtons = timelineRef.current?.querySelectorAll<HTMLButtonElement>('[data-block-id]')
   246→    if (!blockButtons || blockButtons.length === 0) return
   247→
   248→    const currentIndex = Array.from(blockButtons).findIndex(
   249→      (btn) => btn === document.activeElement
   250→    )
   251→
   252→    if (currentIndex === -1) return
   253→
   254→    let nextIndex = currentIndex
   255→    if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
   256→      e.preventDefault()
   257→      nextIndex = (currentIndex + 1) % blockButtons.length
   258→    } else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
   259→      e.preventDefault()
   260→      nextIndex = (currentIndex - 1 + blockButtons.length) % blockButtons.length
   261→    } else {
   262→      return
   263→    }
   264→
   265→    blockButtons[nextIndex]?.focus()
   266→  }, [])
   267→
   268→  // Get conflicts for a block
   269→  const getBlockConflicts = (blockId: string): SimConflict[] => {
   270→    return conflicts.filter((c) => c.blockId === blockId && !c.resolved)
   271→  }
   272→
   273→  // Current time indicator
   274→  const currentHour = new Date().getHours() + new Date().getMinutes() / 60
   275→  const nowPct = (currentHour / 24) * 100
   276→
   277→  if (dayBlocks.length === 0) {
   278→    return (
   279→      <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   280→        <Clock className="h-10 w-10 mb-3 opacity-40" />
   281→        <p className="text-sm font-medium">No blocks scheduled for this day</p>
   282→        <p className="text-xs mt-1">Run AI optimization or manually add blocks</p>
   283→      </div>
   284→    )
   285→  }
   286→
   287→  const timelineAriaLabel = `Block timeline for ${selectedDate}: ${dayBlocks.length} blocks across ${lanes.length} lanes`
   288→
   289→  return (
   290→    <div className="w-full overflow-x-auto" role="img" aria-label={timelineAriaLabel} ref={timelineRef} onKeyDown={handleTimelineKeyDown}>
   291→      <div className="min-w-[800px]">
   292→        {/* Time axis header */}
   293→        <div
   294→          className="relative border-b border-border mb-0"
   295→          style={{ height: HEADER_HEIGHT }}
   296→        >
   297→          {Array.from({ length: TOTAL_HOURS + 1 }, (_, i) => (
   298→            <div
   299→              key={`tick-${i}`}
   300→              className="absolute top-0 flex flex-col items-center"
   301→              style={{ left: `${(i / TOTAL_HOURS) * 100}%`, transform: 'translateX(-50%)' }}
   302→            >
   303→              <span className="text-[10px] text-muted-foreground font-mono mt-1">
   304→                {i.toString().padStart(2, '0')}:00
   305→              </span>
   306→              <div className="w-px h-2 bg-border mt-0.5" />
   307→            </div>
   308→          ))}
   309→        </div>
   310→
   311→        {/* Lanes */}
   312→        <div className="relative">
   313→          {/* Grid lines */}
   314→          <div
   315→            className="absolute inset-0 pointer-events-none"
   316→            style={{ height: lanes.length * (LANE_HEIGHT + LANE_GAP) + LANE_GAP }}
   317→          >
   318→            {Array.from({ length: TOTAL_HOURS + 1 }, (_, i) => (
   319→              <div
   320→                key={`grid-${i}`}
   321→                className="absolute top-0 bottom-0 w-px bg-border/40"
   322→                style={{ left: `${(i / TOTAL_HOURS) * 100}%` }}
   323→              />
   324→            ))}
   325→            {/* 15-min snap guides while dragging — red when the slot overlaps another block */}
   326→            {drag?.active && (
   327→              <div className="absolute inset-0 opacity-30" aria-hidden="true">
   328→                {Array.from({ length: TOTAL_HOURS * 4 + 1 }, (_, i) => (
   329→                  <div
   330→                    key={`snap-${i}`}
   331→                    className={cn(
   332→                      'absolute top-0 bottom-0 w-px',
   333→                      dragConflicts.length > 0 ? 'bg-red-500/70' : 'bg-[#FF9933]/50',
   334→                    )}
   335→                    style={{ left: `${((i * SNAP_H) / TOTAL_HOURS) * 100}%` }}
   336→                  />
   337→                ))}
   338→              </div>
   339→            )}
   340→            {/* Current time indicator with pulse animation */}
   341→            <motion.div
   342→              className="absolute top-0 bottom-0 w-0.5 bg-red-500 z-10"
   343→              style={{ left: `${nowPct}%` }}
   344→              initial={{ opacity: 0 }}
   345→              animate={{ opacity: 1 }}
   346→              transition={{ duration: 0.5 }}
   347→            >
   348→              <motion.div
   349→                className="absolute -top-1.5 -left-2 w-4 h-4 bg-red-500 rounded-full"
   350→                animate={{
   351→                  scale: [1, 1.3, 1],
   352→                  opacity: [1, 0.6, 1],
   353→                }}
   354→                transition={{
   355→                  duration: 2,
   356→                  repeat: Infinity,
   357→                  ease: 'easeInOut',
   358→                }}
   359→              />
   360→              {/* Current time label */}
   361→              <span className="absolute -top-5 -left-7 text-[9px] font-mono text-red-500 font-medium bg-background/80 px-1 rounded">
   362→                {formatHour(currentHour)}
   363→              </span>
   364→            </motion.div>
   365→          </div>
   366→
   367→          {lanes.map((lane) => (
   368→            <div
   369→              key={`lane-${lane.laneIndex}`}
   370→              data-lane
   371→              className="relative"
   372→              style={{
   373→                height: LANE_HEIGHT,
   374→                marginBottom: LANE_GAP,
   375→                marginTop: lane.laneIndex === 0 ? LANE_GAP : 0,
   376→              }}
   377→            >
   378→              {/* Lane label */}
   379→              <div className="absolute -left-0 top-0 bottom-0 flex items-center">
   380→                <span className="text-[10px] text-muted-foreground/60 font-medium">
   381→                  L{lane.laneIndex + 1}
   382→                </span>
   383→              </div>
   384→
   385→              {lane.blocks.map((block) => {
   386→                const startH = timeToHours(block.startTime)
   387→                const endH = timeToHours(block.endTime)
   388→                const isSelected = selectedBlockId === block.id
   389→                const isHovered = hoveredBlockId === block.id
   390→                const blockConflicts = getBlockConflicts(block.id)
   391→                const hasConflict = blockConflicts.length > 0
   392→                const dept = block.department
   393→                const colors = DEPT_COLORS[dept] || DEPT_COLORS.combined
   394→                const lightColors = DEPT_LIGHT_COLORS[dept] || DEPT_LIGHT_COLORS.combined
   395→                const deptLabel = DEPT_LABELS[dept] || dept.toUpperCase().slice(0, 4)
   396→
   397→                // ---- Drag state for this block ----
   398→                const movable = isMovable(block, !!onMoveBlock)
   399→                const isDragging = drag?.active === true && drag.blockId === block.id
   400→                const shownStartH = isDragging ? drag.origStartH + drag.deltaH : startH
   401→                const shownEndH = isDragging ? shownStartH + drag.durationH : endH
   402→                const leftPct = (shownStartH / TOTAL_HOURS) * 100
   403→                const widthPct = ((shownEndH - shownStartH) / TOTAL_HOURS) * 100
   404→                const origLeftPct = (startH / TOTAL_HOURS) * 100
   405→                const origWidthPct = ((endH - startH) / TOTAL_HOURS) * 100
   406→
   407→                return (
   408→                  <HoverCard key={block.id} open={isDragging ? false : isHovered ? undefined : false}>
   409→                    <HoverCardTrigger asChild>
   410→                      <motion.button
   411→                        data-block-id={block.id}
   412→                        className={cn(
   413→                          'absolute top-1 bottom-1 text-left transition-all duration-150 outline-none group/block',
   414→                          movable ? (isDragging ? 'cursor-grabbing' : 'cursor-grab') : 'cursor-pointer',
   415→                          'focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-1',
   416→                          // Rounded corners and shadows
   417→                          'rounded-lg shadow-sm',
   418→                          // AI recommended: dashed border with teal tint, else solid
   419→                          block.isAiRecommended
   420→                            ? cn(lightColors.bg, 'border-2 border-dashed', lightColors.border)
   421→                            : cn(colors.bg, 'border border-solid', colors.border),
   422→                          isSelected && 'ring-2 ring-offset-1 ring-offset-background ring-[#1a237e] shadow-md',
   423→                          isHovered && !isSelected && 'ring-1 ring-offset-1 ring-offset-background ring-[#3f51b5]/60 shadow-md',
   424→                          isDragging && 'z-20 shadow-xl ring-2 ring-[#FF9933] ring-offset-1 ring-offset-background brightness-110 saturate-150',
   425→                          // Focus effect: dim everything else while a drag is in progress
   426→                          drag?.active && !isDragging && 'opacity-50 saturate-50',
   427→                          // Blocks the dragged block would overlap flash a red ring
   428→                          drag?.active && !isDragging && dragConflicts.includes(block.name) && 'opacity-100 saturate-100 ring-2 ring-red-500 shadow-[0_0_10px_rgba(239,68,68,0.5)]',
   429→                        )}
   430→                        style={{
   431→                          left: `${leftPct}%`,
   432→                          width: `${widthPct}%`,
   433→                          touchAction: movable ? 'none' : undefined,
   434→                        }}
   435→                        whileHover={isDragging ? undefined : {
   436→                          boxShadow: '0 4px 12px rgba(0,0,0,0.15)',
   437→                          y: -1,
   438→                        }}
   439→                        animate={isDragging ? { scale: 1.02 } : { scale: 1 }}
   440→                        transition={{ type: 'spring', stiffness: 400, damping: 25 }}
   441→                        onClick={() => {
   442→                          // A completed drag must not also toggle selection
   443→                          if (suppressClickRef.current) {
   444→                            suppressClickRef.current = false
   445→                            return
   446→                          }
   447→                          onSelectBlock(isSelected ? null : block.id)
   448→                        }}
   449→                        onMouseEnter={() => setHoveredBlockId(block.id)}
   450→                        onMouseLeave={() => setHoveredBlockId(null)}
   451→                        onPointerDown={movable ? (e) => handleDragPointerDown(e, block) : undefined}
   452→                        onPointerMove={movable ? handleDragPointerMove : undefined}
   453→                        onPointerUp={movable ? handleDragPointerUp : undefined}
   454→                        onPointerCancel={movable ? handleDragPointerCancel : undefined}
   455→                        onKeyDown={movable ? (e) => handleBlockKeyDown(e, block) : undefined}
   456→                        aria-label={`${block.name}, ${block.department}, ${formatHour(shownStartH)} to ${formatHour(shownEndH)}${hasConflict ? ', has conflicts' : ''}${movable ? ', drag or press Shift with arrow keys to reschedule' : ''}`}
   457→                        layout={false}
   458→                      >
   459→                        {/* Drag handle indicator — persistent on movable blocks, hover-only otherwise */}
   460→                        <div className={cn(
   461→                          'absolute left-0 top-0 bottom-0 w-4 flex items-center justify-center transition-opacity duration-150 pointer-events-none',
   462→                          movable ? 'opacity-35 group-hover/block:opacity-90' : 'opacity-0 group-hover/block:opacity-60',
   463→                        )}>
   464→                          <GripVertical className={cn(
   465→                            'h-3 w-3',
   466→                            block.isAiRecommended ? lightColors.text : 'text-white/60',
   467→                          )} />
   468→                        </div>
   469→
   470→                        {/* Block content */}
   471→                        <div className={cn('flex items-center gap-1 h-full overflow-hidden', movable ? 'pl-4 pr-2' : 'px-2')}>
   472→                          {/* Department label */}
   473→                          <span className={cn(
