   157→  const [currentStatus, setCurrentStatus] = useState<MaintenanceStatus>('pending')

   159→  const [newComment, setNewComment] = useState('')
   160→  const [initialized, setInitialized] = useState(false)
   161→
   162→  // Initialize state when request changes
   163→  const initRequest = useCallback(() => {
   164→    if (request && !initialized) {
   165→      setCurrentStatus(request.status as MaintenanceStatus)
   166→      setComments(getInitialComments(request))
   167→      setInitialized(true)
   168→    }
   169→    if (!request) {
   170→      setInitialized(false)
   171→    }
   172→  }, [request, initialized])
   173→
   174→  // Reset when request changes
   175→  if (request && !initialized) {
   176→    initRequest()
   177→  }
   178→
   179→  if (!request) return null
   180→
   181→  const dept = DEPT_CONFIG[request.department]
   182→  const assignedBlock = request.blockId ? blocks.find((b) => b.id === request.blockId) : null
   183→
   184→  const formatDuration = (mins: number) => {
   185→    if (mins >= 60) {
   186→      const h = Math.floor(mins / 60)
   187→      const m = mins % 60
   188→      return m > 0 ? `${h}h ${m}m` : `${h}h`
   189→    }
   190→    return `${mins}m`
   191→  }
   192→
   193→  const formatDate = (dateStr: string) => {
   194→    return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   195→  }
   196→
   197→  const formatTimestamp = (dateStr: string) => {
   198→    return new Date(dateStr).toLocaleString('en-IN', {
   199→      day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit', hour12: true,
   200→    })
   201→  }
   202→
   203→  // Timeline step index
   204→  const stepIndex = getStepIndex(currentStatus)
   205→  const isRejected = currentStatus === 'rejected'
   206→
   207→  // Handle action button clicks
   208→  const handleScorePriority = () => {

   210→    setComments((prev) => [
   211→      ...prev,
   212→      {
   213→        id: `c-${Date.now()}`,

   215→        initials: 'AI',
   216→        timestamp: new Date().toISOString(),
   217→        message: 'Priority scoring completed. Score calculated based on severity, safety risk, asset criticality, and traffic impact factors.',
   218→      },
   219→    ])
   220→  }
   221→
   222→  const handleAssignDepartment = () => {
   223→    setCurrentStatus('assigned')
   224→    setComments((prev) => [
   225→      ...prev,
   226→      {
   227→        id: `c-${Date.now()}`,
   228→        author: 'Debarshi',
   229→        initials: 'DB',
   230→        timestamp: new Date().toISOString(),
   231→        message: `Assigned to ${dept.label} department. Block window will be proposed by AI optimization engine.`,
   232→      },
   233→    ])
   234→  }
   235→
   236→  const handleMarkVerified = () => {

   238→    setComments((prev) => [
   239→      ...prev,
   240→      {
   241→        id: `c-${Date.now()}`,

   243→        initials: 'RP',

   247→    ])
   248→  }
   249→
   250→  const handleAddComment = () => {

   255→        id: `c-${Date.now()}`,

   264→
   265→  // Action buttons based on status

   278→          </div>
   279→        </SheetHeader>
   280→
   281→        <div className="p-6 space-y-6">

   285→          </div>
   286→
   287→          {/* Department & Category */}
   288→          <div className="flex items-center gap-3">
   289→            <Badge
   290→              variant="outline"

   296→            </Badge>

   313→
   314→              {STATUS_STEPS.map((step, idx) => {

   319→

   356→                          <span className="text-[10px] text-success font-medium">Done</span>

   362→                      <div className="flex items-center gap-2 mt-0.5">

   364→                          {step.actor}

   373→                          </span>
   374→                        )}
   375→                      </div>
   376→                    </div>
   377→                  </motion.div>
   378→                )
   379→              })}
   380→
   381→              {/* Rejected step (shown only when rejected) */}
   382→              {isRejected && (
   383→                <motion.div
   384→                  initial={{ opacity: 0, x: -12 }}
   385→                  animate={{ opacity: 1, x: 0 }}
   386→                  transition={{ duration: 0.25, delay: 0.4 }}
   387→                  className="relative flex items-start gap-3"
   388→                >
   389→                  <div className="relative z-10 mt-0.5">
   390→                    <div className="w-8 h-8 rounded-full bg-destructive/10 flex items-center justify-center">
   391→                      <XCircle className="w-4 h-4 text-destructive" />
   392→                    </div>
   393→                  </div>
   394→                  <div className="flex-1 min-w-0 pt-0.5">
   395→                    <span className="text-sm font-medium text-destructive">Rejected</span>
   396→                    <p className="text-xs text-muted-foreground mt-0.5">
   397→                      Request was rejected — {formatTimestamp(new Date().toISOString())}
   398→                    </p>
   399→                  </div>
   400→                </motion.div>
   401→              )}
   402→            </div>
   403→          </div>
   404→
   405→          <Separator style={{ backgroundColor: '#D7DEE6' }} />
   406→
   407→          {/* Priority Score */}
   408→          <div>
   409→            <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider mb-3 flex items-center gap-1.5">
   410→              <Gauge className="w-3.5 h-3.5" />
   411→              Priority Score
   412→            </h4>
   413→            <div className="flex items-center gap-3 mb-4">
   414→              <PriorityScore
   415→                score={request.priority}
   416→                severity={request.severity}
   417→                safetyRisk={request.safetyRisk}
   418→                assetCriticality={request.assetCriticality}
   419→                trafficImpact={request.trafficImpact}
   420→                showLabel={false}
   421→                size="md"
   422→              />
   423→              <span className="text-2xl font-bold tabular-nums" style={{
   424→                color: request.priority >= 90 ? '#DC2626' : request.priority >= 75 ? '#EA580C' : request.priority >= 50 ? '#D97706' : '#16A34A'
   425→              }}>
   426→                {request.priority}
   427→              </span>
   428→              <span className="text-xs text-muted-foreground/60">/100</span>
   429→            </div>
   430→            <div className="space-y-2.5 bg-muted/50 rounded-lg p-3">
   431→              <FactorBar label="Severity" value={request.severity} />
   432→              <FactorBar label="Safety Risk" value={request.safetyRisk} />
   433→              <FactorBar label="Asset Criticality" value={request.assetCriticality} />
   434→              <FactorBar label="Traffic Impact" value={request.trafficImpact} />
   435→            </div>
   436→          </div>
   437→
   438→          <Separator style={{ backgroundColor: '#D7DEE6' }} />
   439→
   440→          {/* Details */}
   441→          <div>
   442→            <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider mb-3 flex items-center gap-1.5">
   443→              <MapPin className="w-3.5 h-3.5" />
   444→              Details
   445→            </h4>
   446→            <div className="bg-muted/50 rounded-lg p-3 space-y-0.5">

   448→                <span className="font-mono">{request.section}</span>
   449→              </InfoRow>
   450→              <InfoRow label="From">
   451→                {request.stationFrom}
   452→              </InfoRow>
   453→              <InfoRow label="To">
   454→                {request.stationTo}
   455→              </InfoRow>

   465→              {request.isOverdue && (
   466→                <InfoRow label="Overdue">
   467→                  <span className="text-red-600 font-medium flex items-center gap-1">
   468→                    <AlertTriangle className="w-3 h-3" />
   469→                    Yes
   470→                  </span>
   471→                </InfoRow>

   473→              <InfoRow label="Created By">
   474→                <span className="font-mono text-xs">{request.createdBy}</span>

   476→            </div>
   477→          </div>
   478→
   479→          {/* AI Recommendation */}
   480→          {assignedBlock && (
   481→            <>
   482→              <Separator style={{ backgroundColor: '#D7DEE6' }} />
   483→              <div>
   484→                <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wider mb-3 flex items-center gap-1.5">
   485→                  <Brain className="w-3.5 h-3.5" />
   486→                  AI Recommendation
   487→                </h4>
   488→                <div className="bg-[#e8eaf6] border border-[#9fa8da] rounded-lg p-3 space-y-2">
   489→                  <div className="flex items-center justify-between">
   490→                    <span className="text-sm font-medium text-[#1a237e]">{assignedBlock.name}</span>
   491→                    <Badge variant="outline" className="rounded-full text-[10px] px-1.5 bg-teal-100 text-[#0d47a1] border-[#5c6bc0]">
   492→                      {Math.round(assignedBlock.aiConfidence * 100)}% confidence

   502→                </div>

   507→          {/* Risk Factors Summary */}
