    59→// Circular progress indicator component
    60→function CircularProgress({ value, size = 40, strokeWidth = 3 }: { value: number; size?: number; strokeWidth?: number }) {
    61→  const radius = (size - strokeWidth) / 2
    62→  const circumference = 2 * Math.PI * radius
    63→  const progress = ((100 - value) / 100) * circumference
    64→
    65→  const color = value >= 80 ? 'text-emerald-500' : value >= 60 ? 'text-amber-500' : 'text-red-500'
    66→  const strokeColor = value >= 80 ? '#10b981' : value >= 60 ? '#f59e0b' : '#ef4444'
    67→  const bgColor = value >= 80 ? '#10b98120' : value >= 60 ? '#f59e0b20' : '#ef444420'
    68→
    69→  return (
    70→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
    71→      <svg width={size} height={size} className="-rotate-90">
    72→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={bgColor} strokeWidth={strokeWidth} />
    73→        <circle
    74→          cx={size / 2}
    75→          cy={size / 2}
    76→          r={radius}
    77→          fill="none"
    78→          stroke={strokeColor}
    79→          strokeWidth={strokeWidth}
    80→          strokeDasharray={circumference}
    81→          strokeDashoffset={progress}
    82→          strokeLinecap="round"
    83→          className="transition-all duration-700 ease-out"
    84→        />
    85→      </svg>
    86→      <span className={`absolute text-[10px] font-bold ${color}`}>{value}</span>
    87→    </div>
    88→  )
    89→}

    91→function exportPlanAsJSON(plan: typeof plans[0], planBlocks: typeof blocks, planRequests: typeof maintenanceRequests) {
    92→  const exportData = {
    93→    plan: { id: plan.id, name: plan.name, type: plan.type, startDate: plan.startDate, endDate: plan.endDate, status: plan.status, version: plan.version },
    94→    blocks: planBlocks.map(b => ({ id: b.id, name: b.name, section: b.section, stationFrom: b.stationFrom, stationTo: b.stationTo, startTime: b.startTime, endTime: b.endTime, duration: b.duration, department: b.department, line: b.line, isAiRecommended: b.isAiRecommended, confidence: b.aiConfidence })),
    95→    maintenanceRequests: planRequests.map(r => ({ id: r.id, title: r.title, department: r.department, priority: r.priority, severity: r.severity, status: r.status })),
    96→    exportedAt: new Date().toISOString(),
    97→  }
    98→  const blob = new Blob([JSON.stringify(exportData, null, 2)], { type: 'application/json' })
    99→  const url = URL.createObjectURL(blob)
   100→  const a = document.createElement('a')
   101→  a.href = url
   102→  a.download = `plan-${plan.id}-${new Date().toISOString().split('T')[0]}.json`
   103→  a.click()
   104→  URL.revokeObjectURL(url)
   105→  toast.success('Plan exported', { description: `${plan.name} downloaded as JSON` })
   106→}

   108→function exportPlanAsCSV(plan: typeof plans[0], planBlocks: typeof blocks) {
   109→  const headers = ['Block ID', 'Name', 'Section', 'From', 'To', 'Start', 'End', 'Duration (min)', 'Department', 'Line', 'AI Recommended', 'Confidence']
   110→  const rows = planBlocks.map(b => [b.id, b.name, b.section, b.stationFrom, b.stationTo, b.startTime, b.endTime, b.duration, b.department, b.line, b.isAiRecommended ? 'Yes' : 'No', b.aiConfidence.toFixed(2)])
   111→  const csv = [headers.join(','), ...rows.map(r => r.join(','))].join('\n')
   112→  const blob = new Blob([csv], { type: 'text/csv' })
   113→  const url = URL.createObjectURL(blob)
   114→  const a = document.createElement('a')
   115→  a.href = url
   116→  a.download = `plan-${plan.id}-${new Date().toISOString().split('T')[0]}.csv`
   117→  a.click()
   118→  URL.revokeObjectURL(url)
   119→  toast.success('Plan exported', { description: `${plan.name} downloaded as CSV` })
   120→}
